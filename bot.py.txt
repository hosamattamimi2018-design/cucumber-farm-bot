def convert_arabic_numbers(text: str) -> str:
    arabic_digits = "٠١٢٣٤٥٦٧٨٩"
    english_digits = "0123456789"
    table = str.maketrans(arabic_digits, english_digits)
    return text.translate(table)


def format_number(number):
    try:
        return f"{float(number):,.2f}"
    except Exception:
        return "0.00"


def format_ar_pdf(text: str) -> str:
    reshaped_text = arabic_reshaper.reshape(str(text))
    return get_display(reshaped_text)


def get_now():
    return datetime.datetime.now()


def get_date_string():
    return get_now().strftime("%Y-%m-%d %H:%M:%S")


def get_today_string():
    return get_now().strftime("%Y-%m-%d")


def generate_transaction_id():
    now = get_now().strftime("%Y%m%d%H%M%S")
    short_id = uuid.uuid4().hex[:6].upper()
    return f"CUC-{now}-{short_id}"


def detect_currency(text):
    text = convert_arabic_numbers(text).lower()
    if any(word in text for word in ["سعودي", "سعوديه", "ريال سعودي", "ر.س", "رس"]):
        return "ريال سعودي"
    return "ريال يمني"


def extract_numbers(text):
    text = convert_arabic_numbers(text)
    numbers = re.findall(r"\d+(?:\.\d+)?", text)
    return [float(n) for n in numbers]


def extract_amount(text):
    numbers = extract_numbers(text)
    if not numbers:
        return 0.0
    return numbers[-1]


def extract_quantity(text):
    text = convert_arabic_numbers(text).lower()
    match = re.search(r"(\d+(?:\.\d+)?)\s*(?:كيلو|كجم|كغ|kg)", text)
    if match:
        return float(match.group(1))
    return 0.0


def extract_price(text):
    text = convert_arabic_numbers(text).lower()
    patterns = [
        r"بسعر\s*(\d+(?:\.\d+)?)",
        r"سعر\s*(\d+(?:\.\d+)?)",
        r"السعر\s*(\d+(?:\.\d+)?)"
    ]
    for pattern in patterns:
        match = re.search(pattern, text)
        if match:
            return float(match.group(1))
    return 0.0

def calculate_balances():
    records = sheet.get_all_values()
    balances = {
        "ريال سعودي": 0.0,
        "ريال يمني": 0.0
    }

    if len(records) <= 1:
        return balances

    for row in records[1:]:
        if len(row) < 6:
            continue

        transaction_type = row[1].strip()
        currency = row[5].strip()

        try:
            amount = float(row[4].replace(",", "").strip())
        except Exception:
            continue

        if transaction_type in ["إنتاج", "آجل / دين"]:
            continue

        if currency in balances:
            balances[currency] += amount

    return balances


def get_balance_after_transaction(currency, amount, transaction_type):
    balances = calculate_balances()
    current_balance = balances.get(currency, 0.0)
    if transaction_type in ["إنتاج", "آجل / دين"]:
        return current_balance
    return current_balance + amount

):

    return {

        "balance": balance
    }

    lower_text = convert_arabic_numbers(text).lower()
    categories = {

        "معدات وأدوات": ["معدات", "أداة", "ادوات", "مقص", "ميزان"]
    }
    for category, keywords in categories.items():
        if any(word in lower_text for word in keywords):
            return category
    return "مصروفات عامة"


def analyze_message(text):
    lower_text = convert_arabic_numbers(text).lower()
    currency = detect_currency(text)
    quantity = extract_quantity(text)
    price = extract_price(text)

    if any(word in lower_text for word in ["انتاج", "إنتاج", "حصد", "حصاد"]):
        if quantity <= 0:
            quantity = extract_amount(text)
        return {
            "type": "إنتاج",
            "category": "إنتاج محصول الخيار",
            "amount": quantity,
            "currency": "كيلو",
            "operation": "إنتاج",
            "quantity": quantity,
            "details": text,
            "explanation": f"تم تسجيل إنتاج {quantity} كيلو خيار"
        }

    if any(word in lower_text for word in ["بيع", "بعت", "مبيعات", "ايراد", "إيراد", "دخل", "محصول"]):
        if quantity > 0 and price > 0:
            amount = quantity * price
            explanation = f"بيع {quantity:,.2f} كيلو خيار بسعر {price:,.2f} للكيلو"
        else:
            amount = extract_amount(text)
            explanation = "إيراد من بيع محصول الخيار"
        return {
            "type": "إيراد",
            "category": "مبيعات محصول الخيار",
            "amount": abs(amount),
            "currency": currency,
            "operation": "بيع",
            "quantity": quantity,
            "details": text,
            "explanation": explanation
        }

    if any(word in lower_text for word in ["دين", "آجل", "اجل", "مؤجل", "مؤجلة"]):
        amount = extract_amount(text)
        return {
            "type": "آجل / دين",
            "category": "ذمم وديون",
            "amount": abs(amount),
            "currency": currency,
            "operation": "آجل",
            "quantity": 0,
            "details": text,
            "explanation": "عملية آجل أو دين"
        }

    amount = extract_amount(text)
    return {
        "type": "مصروف",
        "category": detect_expense_category(text),
        "amount": -abs(amount),
        "currency": currency,
        "operation": "صرف",
        "quantity": 0,
        "details": text,
        "explanation": "مصروف لمشروع زراعة الخيار"
    }

        transactions_count += 1

            amount = float(row[4].replace(",", ""))

        "sar_income": sar_income,
        "sar_expense": sar_expense,
        "yer_income": yer_income,

# معالجة الصرافة
# =========================================================

async def handle_exchange(update: Update, context: ContextTypes.DEFAULT_TYPE, text: str):
    norm_text = convert_arabic_numbers(text)
    pattern = r"^صرف\s+(\d+(?:\.\d+)?)\s*(?:سعودي|ر\.س|رس)?\s*بسعر\s*(\d+(?:\.\d+)?)"
    match = re.search(pattern, norm_text, re.IGNORECASE)

    if not match:
        return False

    sar_amount = float(match.group(1))
    rate = float(match.group(2))
    yer_amount = sar_amount * rate

    transaction_id = generate_transaction_id()
    date_string = get_date_string()
    balances = calculate_balances()
    user_name = update.effective_user.full_name or update.effective_user.first_name or "مستخدم"

    new_sar_balance = balances["ريال سعودي"] - sar_amount
    new_yer_balance = balances["ريال يمني"] + yer_amount

    sheet.append_row([
        date_string, "تحويل عملة", "صرافة",
        f"صرف {sar_amount} ريال سعودي بسعر {rate}",
        -sar_amount, "ريال سعودي", "صرافة",
        new_sar_balance, f"تحويل سعودي إلى يمني بسعر {rate}",
        transaction_id, user_name
    ])

    sheet.append_row([
        date_string, "تحويل عملة", "صرافة",
        f"استلام {yer_amount} ريال يمني",
        yer_amount, "ريال يمني", "صرافة",
        new_yer_balance, f"مقابل {sar_amount} ريال سعودي بسعر {rate}",
        transaction_id, user_name
    ])

    reply = (
        "✅ *تمت عملية الصرافة بنجاح*\n\n"
        f"🔻 تم خصم: {format_number(sar_amount)} ر.س\n"
        f"🔺 تم إضافة: {format_number(yer_amount)} ر.ي\n"
        f"💱 سعر الصرف: {format_number(rate)}\n"
        f"👤 المنفذ: {user_name}\n"
        f"🆔 العملية: `{transaction_id}`"
    )

    await update.message.reply_text(reply, parse_mode="Markdown")

    # إشعار للمدير
    admin_alert = f"🔔 *إشعار صرافة جديدة*\n\n" + reply
    await notify_admins(context, admin_alert, update.effective_chat.id)

    return True


# =========================================================
# استقبال ومعالجة الرسائل
# =========================================================

async def handle_message(update: Update, context: ContextTypes.DEFAULT_TYPE):
    if not update.message or not update.message.text:
        return

    raw_text = update.message.text.strip()
    if not raw_text:
        return

    if raw_text.startswith("صرف"):
        if await handle_exchange(update, context, raw_text):
            return

    transaction = analyze_message(raw_text)

    if transaction["amount"] == 0 and transaction["type"] != "إنتاج":
        await update.message.reply_text(
            "⚠️ *لم أتمكن من معرفة المبلغ.*\n\n"
            "مثال:\n"
            "`سماد 25000 يمني`\n"
            "`بيع خيار 500 كيلو بسعر 120 سعودي`",
            parse_mode="Markdown"
        )
        return

    user_name = update.effective_user.full_name or update.effective_user.first_name or "مستخدم"

    result = add_transaction(
        transaction_type=transaction["type"],
        category=transaction["category"],
        details=transaction["details"],
        amount=transaction["amount"],
        currency=transaction["currency"],
        operation=transaction["operation"],
        explanation=transaction["explanation"],
        user_name=user_name
    )

    icons = {"إيراد": "🟢", "مصروف": "🔴", "إنتاج": "🥒", "آجل / دين": "🟡"}
    icon = icons.get(transaction["type"], "✅")

    if transaction["type"] == "إنتاج":
        reply = (
            f"{icon} *تم تسجيل الإنتاج بنجاح*\n\n"
            f"📅 التاريخ: {result['date']}\n"
            f"⚖️ الكمية: {format_number(transaction['amount'])} كيلو\n"
            f"📂 الفئة: {transaction['category']}\n"
            f"👤 المسجل: {user_name}\n"
            f"📝 البيان: {raw_text}\n"
            f"🆔 المعرف: `{result['id']}`"
        )
    else:
        reply = (
            f"{icon} *تم تسجيل العملية بنجاح*\n\n"
            f"📅 التاريخ: {result['date']}\n"
            f"🏷️ النوع: {transaction['type']}\n"
            f"📂 الفئة: {transaction['category']}\n"
            f"💰 المبلغ: {format_number(abs(transaction['amount']))} {transaction['currency']}\n"
        )
        if transaction["quantity"] > 0:
            reply += f"⚖️ الكمية: {format_number(transaction['quantity'])} كيلو\n"

        reply += (
            f"🏦 الرصيد: {format_number(result['balance'])} {transaction['currency']}\n"
            f"👤 المسجل: {user_name}\n"
            f"📝 البيان: {raw_text}\n"
            f"🆔 المعرف: `{result['id']}`"
        )

    await update.message.reply_text(reply, parse_mode="Markdown")

    # إرسال إشعار فوري للمدير
    admin_alert = f"🔔 *إشعار عملية جديدة بالمحمية*\n\n" + reply
    await notify_admins(context, admin_alert, update.effective_chat.id)

        "help_exchange": "💱 *عملية صرافة*\n\nمثال:\n`صرف 200 سعودي بسعر 535`"

        await query.message.reply_text(help_messages[action], parse_mode="Markdown")

        balances = calculate_balances()

            f"🏦 *أرصدة الكاش الحالية*\n\n🇸🇦 سعودي: {format_number(balances['ريال سعودي'])} ر.س\n🇾🇪 يمني: {format_number(balances['ريال يمني'])} ر.ي",
            parse_mode="Markdown"
        )
        return

    if action == "cmd_today":
        report = get_today_report()
        await query.message.reply_text(
            f"📊 *تقرير اليوم — {get_today_string()}*\n\n⚖️ الإنتاج: {format_number(report['production'])} كيلو\n\n"
            f"🇸🇦 *الريال السعودي*\n🟢 الإيرادات: {format_number(report['sar_income'])} ر.س\n🔴 المصروفات: {format_number(report['sar_expense'])} ر.س\n📈 الصافي: {format_number(report['sar_income'] - report['sar_expense'])} ر.س\n\n"
            f"🇾🇪 *الريال اليمني*\n🟢 الإيرادات: {format_number(report['yer_income'])} ر.ي\n🔴 المصروفات: {format_number(report['yer_expense'])} ر.ي\n📈 الصافي: {format_number(report['yer_income'] - report['yer_expense'])} ر.ي\n\n"
            f"📋 عدد العمليات: {report['count']}",
            parse_mode="Markdown"
        )
        return

    if action == "cmd_month":
        report = get_month_report()
        month = get_now().strftime("%Y-%m")
        await query.message.reply_text(
            f"📅 *التقرير الشهري — {month}*\n\n⚖️ إجمالي الإنتاج: {format_number(report['production'])} كيلو\n\n"
            f"🇸🇦 سعودي\n🟢 الإيرادات: {format_number(report['sar_income'])} ر.س\n🔴 المصروفات: {format_number(report['sar_expense'])} ر.س\n📈 الصافي: {format_number(report['sar_income'] - report['sar_expense'])} ر.س\n\n"
            f"🇾🇪 يمني\n🟢 الإيرادات: {format_number(report['yer_income'])} ر.ي\n🔴 المصروفات: {format_number(report['yer_expense'])} ر.ي\n📈 الصافي: {format_number(report['yer_income'] - report['yer_expense'])} ر.ي\n\n"
            f"📋 عدد العمليات: {report['count']}",
            parse_mode="Markdown"
        )
        return

    if action == "cmd_profits":
        report = get_profit_report()
        await query.message.reply_text(
            f"📈 *التقرير المالي الكامل للمشروع*\n\n🥒 إجمالي الإنتاج المسجل: {format_number(report['production'])} كيلو\n\n"
            f"🇸🇦 *ريال سعودي*\n🟢 إجمالي الإيرادات: {format_number(report['sar_income'])} ر.س\n🔴 إجمالي المصروفات: {format_number(report['sar_expense'])} ر.س\n💰 صافي الربح: {format_number(report['sar_profit'])} ر.س\n\n"
            f"🇾🇪 *ريال يمني*\n🟢 إجمالي الإيرادات: {format_number(report['yer_income'])} ر.ي\n🔴 إجمالي المصروفات: {format_number(report['yer_expense'])} ر.ي\n💰 صافي الربح: {format_number(report['yer_profit'])} ر.ي",
            parse_mode="Markdown"
        )
        return

    if action == "cmd_debts":
        sar_debt, yer_debt = get_debts()
        await query.message.reply_text(
            f"📑 *إجمالي الديون والعمليات الآجلة*\n\n🇸🇦 سعودي: {format_number(sar_debt)} ر.س\n🇾🇪 يمني: {format_number(yer_debt)} ر.ي",
            parse_mode="Markdown"
        )
        return

    if action == "cmd_pdf":
        await query.message.reply_text("⏳ جارٍ إعداد وتوليد تقرير PDF...")
        try:
            pdf_file = generate_pdf_report()
            today_str = get_today_string()
            await query.message.reply_document(
                document=pdf_file,
                filename=f"Cucumber_Farm_Report_{today_str}.pdf",
                caption=f"📄 *تقرير مشروع زراعة الخيار*\n📅 التاريخ: {today_str}",
                parse_mode="Markdown"
            )
        except Exception as e:
            await query.message.reply_text(f"❌ تعذر إنشاء ملف الـ PDF: {str(e)}")
        return

    if action == "cmd_undo":
        records = sheet.get_all_values()
        if len(records) <= 1:
            await query.message.reply_text("⚠️ لا توجد عمليات لحذفها.")
            return

        last_row = records[-1]
        sheet.delete_rows(len(records))
        user_name = update.effective_user.full_name or update.effective_user.first_name or "مستخدم"
        
        reply = (
            f"🔄 *تم حذف آخر عملية بنجاح*\n\n"
            f"📅 التاريخ: {last_row[0]}\n"
            f"🏷️ النوع: {last_row[1]}\n"
            f"📂 الفئة: {last_row[2]}\n"
            f"💰 المبلغ: {last_row[4]} {last_row[5]}\n"
            f"👤 بواسطة: {user_name}"
        )
        await query.message.reply_text(reply, parse_mode="Markdown")

        # إشعار المدير بالحذف
        admin_alert = f"⚠️ *إشعار حذف عملية من النظام*\n\n" + reply
        await notify_admins(context, admin_alert, update.effective_chat.id)
        return
