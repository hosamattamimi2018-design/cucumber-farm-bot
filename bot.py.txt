def convert_arabic_numbers(text: str) -> str:
    arabic_digits = "٠١٢٣٤٥٦٧٨٩"
    english_digits = "0123456789"
    table = str.maketrans(arabic_digits, english_digits)
    return text.translate(table)

def format_ar_pdf(text: str) -> str:
    reshaped_text = arabic_reshaper.reshape(str(text))
    return get_display(reshaped_text)

def get_now():
    return datetime.datetime.now()

def get_date_string():
    return get_now().strftime("%Y-%m-%d %H:%M:%S")

def get_today_string():
    return get_now().strftime("%Y-%m-%d")

def generate_transaction_id():
    now = get_now().strftime("%Y%m%d%H%M%S")
    short_id = uuid.uuid4().hex[:6].upper()
    return f"CUC-{now}-{short_id}"

def detect_currency(text):
    text = convert_arabic_numbers(text).lower()
    if any(word in text for word in ["سعودي", "سعوديه", "ريال سعودي", "ر.س", "رس"]):
        return "ريال سعودي"
    return "ريال يمني"

def extract_numbers(text):
    text = convert_arabic_numbers(text)
    numbers = re.findall(r"\d+(?:\.\d+)?", text)
    return [float(n) for n in numbers]

def extract_amount(text):
    numbers = extract_numbers(text)
    if not numbers:
        return 0.0
    return numbers[-1]

def extract_quantity(text):
    text = convert_arabic_numbers(text).lower()
    match = re.search(r"(\d+(?:\.\d+)?)\s*(?:كيلو|كجم|كغ|kg)", text)
    if match:
        return float(match.group(1))
    return 0.0

def calculate_balances():
    balances = {"ريال سعودي": 0.0, "ريال يمني": 0.0}
    if not sheet:
        return balances

    try:
        records = sheet.get_all_values()
        if len(records) <= 1:
            return balances

        for row in records[1:]:
            if len(row) < 6:
                continue

            transaction_type = row[1].strip()
            currency = row[5].strip()

            try:
                amount = float(row[4].replace(",", "").strip())
            except Exception:
                continue

            if transaction_type in ["إنتاج", "آجل / دين"]:
                continue

            if currency in balances:
                balances[currency] += amount
    except Exception as e:
        print(f"Error calculate_balances: {e}")

    return balances

def get_balance_after_transaction(currency, amount, transaction_type):
    balances = calculate_balances()
    current_balance = balances.get(currency, 0.0)
    if transaction_type in ["إنتاج", "آجل / دين"]:
        return current_balance
    return current_balance + amount

def parse_invoice_image(image_bytes: bytes) -> dict:
    try:
        # استخدام ملف الحساب الخدمي لتوليد OAuth 2.0 Access Token رسمي
        scope = ["https://www.googleapis.com/auth/cloud-platform"]
        sa_creds = ServiceAccountCredentials.from_json_keyfile_name(CREDENTIALS_FILE, scope)
        access_token = sa_creds.get_access_token().access_token

        img = Image.open(io.BytesIO(image_bytes))
        img = ImageOps.exif_transpose(img)

        # ضبط اتجاه الصورة الرأسي
        if img.width > img.height:
            img = img.rotate(-90, expand=True)

        img.thumbnail((1600, 1600))
        buffered = io.BytesIO()
        img.save(buffered, format="JPEG", quality=90)
        img_b64 = base64.b64encode(buffered.getvalue()).decode('utf-8')

        prompt = (
            "أنت مدقق محاسبي لفواتير سوق الخضار والفواكه المركزي.\n"
            "استخرج البيانات التالية بدقة كأرقام صحيحة فقط:\n"
            "1. boxes: مجموع عمود الكمية (عدد الصناديق/السلات).\n"
            "2. gross: الإجمالي العام قبل الخصم.\n"
            "3. net: الرصيد المتبقي الصافي (المطابق للتفقيط المكتوب بالكلمات أسفل الفاتورة).\n\n"
            "إذا تعذر قراءة الصافي: net = gross * 0.97.\n"
            "أخرج الناتج كـ JSON فقط بالصيغة:\n"
            '{"boxes": <number>, "gross": <number>, "net": <number>}'
        )

        url = "https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash:generateContent"
        
        headers = {
            "Content-Type": "application/json",
            "Authorization": f"Bearer {access_token}",
            "x-goog-user-project": sa_creds.project_id
        }

        payload = {
            "contents": [
                {
                    "parts": [
                        {"text": prompt},
                        {
                            "inline_data": {
                                "mime_type": "image/jpeg",
                                "data": img_b64
                            }
                        }
                    ]
                }
            ],
            "generationConfig": {
                "temperature": 0.0,
                "response_mime_type": "application/json"
            }
        }

        res = requests.post(url, headers=headers, json=payload, timeout=35)
        
        if res.status_code != 200:
            return {"error": f"Google Error ({res.status_code}): {res.text}"}

        resp_json = res.json()
        raw_text = resp_json['candidates'][0]['content']['parts'][0]['text'].strip()
        data = json.loads(raw_text)

        boxes = float(data.get("boxes", 0))
        gross = float(data.get("gross", 0))
        net = float(data.get("net", 0))

        if net == 0 and gross > 0:
            net = round(gross * 0.97)

        return {
            "boxes_count": boxes,
            "invoice_net_amount": net,
            "currency": "ريال يمني"
        }

    except Exception as e:
        return {"error": str(e)}

async def handle_photo(update: Update, context: ContextTypes.DEFAULT_TYPE):
    if not update.message or not update.message.photo:
        return

    caption = (update.message.caption or "").strip()
    caption_nums = extract_numbers(caption)

    currency = detect_currency(caption)
    boxes = 0.0
    invoice_net = 0.0

    if len(caption_nums) >= 2:
        boxes = caption_nums[0]
        invoice_net = caption_nums[1]
    else:
        status_msg = await update.message.reply_text("⏳ جاري تدقيق الفاتورة عبر Gemini...")
        try:
            photo_file = await update.message.photo[-1].get_file()
            photo_bytes = await photo_file.download_as_bytearray()
            data = parse_invoice_image(bytes(photo_bytes))
        except Exception as e:
            await status_msg.edit_text(f"❌ {e}")
            return

        if "error" in data:
            await status_msg.edit_text(f"❌ {data['error']}")
            return

        boxes = float(data.get("boxes_count", 0))
        invoice_net = float(data.get("invoice_net_amount", 0))

        if boxes <= 0 or invoice_net <= 0:
            await status_msg.edit_text(
                "⚠️ لم يتم تحديد الأرقام بدقة.\n"
                "أعد إرسالها مع كتابة: عدد الصناديق والمبلغ (مثال: `9 118825`)",
                parse_mode="Markdown"
            )
            return

    packing_cost = boxes * 500.0
    transport_cost = boxes * 500.0
    total_deductions = packing_cost + transport_cost
    final_net = invoice_net - total_deductions

    user_name = update.effective_user.first_name or "مستخدم"

    rev_res = add_transaction("إيراد", "مبيعات خيار", f"فاتورة {boxes:.0f} صندوق", invoice_net, currency, "بيع", "", user_name)
    add_transaction("مصروف", "تغليف وكراتين", f"تكييس {boxes:.0f} صندوق", -packing_cost, currency, "صرف", "", user_name)
    add_transaction("مصروف", "نقل وتسويق", f"نقل {boxes:.0f} صندوق", -transport_cost, currency, "صرف", "", user_name)

    final_balance = rev_res['balance'] - total_deductions

    summary = (
        f"📦 الصناديق: {boxes:,.0f}\n"
        f"💵 صافي الفاتورة: {invoice_net:,.0f}\n"
        f"✂️ التكييس: {packing_cost:,.0f}\n"
        f"🚚 النقل: {transport_cost:,.0f}\n"
        f"💰 الصافي النهائي: {final_net:,.0f}\n"
        f"🏦 الرصيد: {final_balance:,.0f} {currency}"
    )

    if 'status_msg' in locals():
        await status_msg.edit_text(summary)
    else:
        await update.message.reply_text(summary)

    await notify_admins(context, summary, update.effective_chat.id)

async def handle_exchange(update: Update, context: ContextTypes.DEFAULT_TYPE, text: str):
    norm_text = convert_arabic_numbers(text)
    pattern = r"^صرف\s+(\d+(?:\.\d+)?)\s*(?:سعودي|ر\.س|رس)?\s*بسعر\s*(\d+(?:\.\d+)?)"
    match = re.search(pattern, norm_text, re.IGNORECASE)

    if not match:
        return False

    sar_amount = float(match.group(1))
    rate = float(match.group(2))
    yer_amount = sar_amount * rate

    transaction_id = generate_transaction_id()
    date_string = get_date_string()
    balances = calculate_balances()
    user_name = update.effective_user.first_name or "مستخدم"

    new_sar_balance = balances["ريال سعودي"] - sar_amount
    new_yer_balance = balances["ريال يمني"] + yer_amount

    if sheet:
        sheet.append_row([
            date_string, "تحويل عملة", "صرافة",
            f"صرف {sar_amount} ريال سعودي بسعر {rate}",
            -sar_amount, "ريال سعودي", "صرافة",
            new_sar_balance, f"تحويل سعودي إلى يمني بسعر {rate}",
            transaction_id, user_name
        ])

        sheet.append_row([
            date_string, "تحويل عملة", "صرافة",
            f"استلام {yer_amount} ريال يمني",
            yer_amount, "ريال يمني", "صرافة",
            new_yer_balance, f"مقابل {sar_amount} ريال سعودي بسعر {rate}",
            transaction_id, user_name
        ])

    reply = (
        f"💱 صرافة: {sar_amount:,.0f} ر.س\n"
        f"💵 المستلم: {yer_amount:,.0f} ر.ي\n"
        f"📈 السعر: {rate:,.0f}\n"
        f"🇸🇦 رصيد سعودي: {new_sar_balance:,.0f}\n"
        f"🇾🇪 رصيد يمني: {new_yer_balance:,.0f}"
    )

    await update.message.reply_text(reply)
    await notify_admins(context, reply, update.effective_chat.id)

    return True

async def handle_buttons(update: Update, context: ContextTypes.DEFAULT_TYPE):
    query = update.callback_query
    await query.answer()
    action = query.data

    if action == "cmd_balance":
        balances = calculate_balances()
        await query.message.reply_text(
            f"🇸🇦 سعودي: {balances['ريال سعودي']:,.0f} ر.س\n"
            f"🇾🇪 يمني: {balances['ريال يمني']:,.0f} ر.ي"
        )
    elif action == "cmd_today":
        report = get_today_report()
        await query.message.reply_text(
            f"📊 اليوم: {get_today_string()}\n"
            f"⚖️ إنتاج: {report['production']:,.0f} كجم\n\n"
            f"🇸🇦 صافي سعودي: {(report['sar_income'] - report['sar_expense']):,.0f}\n"
            f"🇾🇪 صافي يمني: {(report['yer_income'] - report['yer_expense']):,.0f}"
        )
    elif action == "cmd_month":
        report = get_month_report()
        month = get_now().strftime("%Y-%m")
        await query.message.reply_text(
            f"📅 شهر: {month}\n"
            f"⚖️ إنتاج: {report['production']:,.0f} كجم\n\n"
            f"🇸🇦 صافي سعودي: {(report['sar_income'] - report['sar_expense']):,.0f}\n"
            f"🇾🇪 صافي يمني: {(report['yer_income'] - report['yer_expense']):,.0f}"
        )
    elif action == "cmd_profits":
        report = get_profit_report()
        await query.message.reply_text(
            f"📈 الأرباح الكاملة للمشروع:\n"
            f"⚖️ إجمالي الإنتاج: {report['production']:,.0f} كجم\n"
            f"🇸🇦 صافي سعودي: {report['sar_profit']:,.0f}\n"
            f"🇾🇪 صافي يمني: {report['yer_profit']:,.0f}"
        )
    elif action == "cmd_debts":
        sar_debt, yer_debt = get_debts()
        await query.message.reply_text(
            f"📑 الديون المسجلة:\n🇸🇦 سعودي: {sar_debt:,.0f}\n🇾🇪 يمني: {yer_debt:,.0f}"
        )
    elif action == "cmd_pdf":
        await query.message.reply_text("⏳ جاري إنشاء تقرير PDF...")
        try:
            pdf_file = generate_pdf_report()
            today_str = get_today_string()
            await query.message.reply_document(
                document=pdf_file,
                filename=f"Report_{today_str}.pdf",
                caption=f"📄 تقرير {today_str}"
            )
        except Exception as e:
            await query.message.reply_text(f"❌ {e}")
    elif action == "cmd_undo":
        if not sheet:
            await query.message.reply_text("⚠️ لم يتم الاتصال بالشيت بعد")
            return
        records = sheet.get_all_values()
        if len(records) <= 1:
            await query.message.reply_text("⚠️ لا توجد عمليات لحذفها")
            return
        last_row = records[-1]
        sheet.delete_rows(len(records))
        reply = f"🔄 تم حذف آخر عملية:\n{last_row[1]} | {last_row[4]} {last_row[5]}"
        await query.message.reply_text(reply)
        await notify_admins(context, reply, update.effective_chat.id)

async def handle_message(update: Update, context: ContextTypes.DEFAULT_TYPE):
    if not update.message or not update.message.text:
        return

    raw_text = update.message.text.strip()
    if not raw_text:
        return

    if raw_text.startswith("صرف"):
        if await handle_exchange(update, context, raw_text):
            return

    if raw_text.startswith("فاتورة") or raw_text.startswith("فاتوره"):
        nums = extract_numbers(raw_text)
        if len(nums) >= 2:
            boxes = nums[0]
            invoice_net = nums[1]
            currency = detect_currency(raw_text)
            packing_cost = boxes * 500.0
            transport_cost = boxes * 500.0
            total_deductions = packing_cost + transport_cost
            final_net = invoice_net - total_deductions
            user_name = update.effective_user.first_name or "مستخدم"

            rev_res = add_transaction("إيراد", "مبيعات خيار", f"فاتورة {boxes:.0f} صندوق", invoice_net, currency, "بيع", "", user_name)
            add_transaction("مصروف", "تغليف وكراتين", f"تكييس {boxes:.0f} صندوق", -packing_cost, currency, "صرف", "", user_name)
            add_transaction("مصروف", "نقل وتسويق", f"نقل {boxes:.0f} صندوق", -transport_cost, currency, "صرف", "", user_name)
            final_balance = rev_res['balance'] - total_deductions

            summary = (
                f"📦 الصناديق: {boxes:,.0f}\n"
                f"💵 صافي الفاتورة: {invoice_net:,.0f}\n"
                f"✂️ التكييس: {packing_cost:,.0f}\n"
                f"🚚 النقل: {transport_cost:,.0f}\n"
                f"💰 الصافي النهائي: {final_net:,.0f}\n"
                f"🏦 الرصيد: {final_balance:,.0f} {currency}"
            )
            await update.message.reply_text(summary)
            await notify_admins(context, summary, update.effective_chat.id)
            return

    transaction = analyze_message(raw_text)
    if transaction["amount"] == 0 and transaction["type"] != "إنتاج":
        return

    user_name = update.effective_user.first_name or "مستخدم"
    result = add_transaction(
        transaction_type=transaction["type"],
        category=transaction["category"],
        details=transaction["details"],
        amount=transaction["amount"],
        currency=transaction["currency"],
        operation=transaction["operation"],
        explanation=transaction["explanation"],
        user_name=user_name
    )

    reply = (
        f"🏷️ النوع: {transaction['type']}\n"
        f"💰 المبلغ: {abs(transaction['amount']):,.0f} {transaction['currency']}\n"
        f"🏦 الرصيد: {result['balance']:,.0f} {transaction['currency']}"
    )
    await update.message.reply_text(reply)
