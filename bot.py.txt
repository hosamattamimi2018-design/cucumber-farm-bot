def convert_arabic_numbers(text: str) -> str:
    arabic_digits = "٠١٢٣٤٥٦٧٨٩"
    english_digits = "0123456789"
    table = str.maketrans(arabic_digits, english_digits)
    return text.translate(table)

def format_ar_pdf(text: str) -> str:
    reshaped_text = arabic_reshaper.reshape(str(text))
    return get_display(reshaped_text)

def get_now():
    return datetime.datetime.now()

def get_date_string():
    return get_now().strftime("%Y-%m-%d %H:%M:%S")

def get_today_string():
    return get_now().strftime("%Y-%m-%d")

def generate_transaction_id():
    now = get_now().strftime("%Y%m%d%H%M%S")
    short_id = uuid.uuid4().hex[:6].upper()
    return f"CUC-{now}-{short_id}"

def detect_currency(text):
    text = convert_arabic_numbers(text).lower()
    if any(word in text for word in ["سعودي", "سعوديه", "ريال سعودي", "ر.س", "رس"]):
        return "ريال سعودي"
    return "ريال يمني"

def extract_numbers(text):
    text = convert_arabic_numbers(text)
    numbers = re.findall(r"\d+(?:\.\d+)?", text)
    return [float(n) for n in numbers]

def extract_amount(text):
    numbers = extract_numbers(text)
    if not numbers:
        return 0.0
    return numbers[-1]

def extract_quantity(text):
    text = convert_arabic_numbers(text).lower()
    match = re.search(r"(\d+(?:\.\d+)?)\s*(?:كيلو|كجم|كغ|kg)", text)
    if match:
        return float(match.group(1))
    return 0.0

def calculate_balances():
    balances = {"ريال سعودي": 0.0, "ريال يمني": 0.0}
    if not sheet:
        return balances

    try:
        records = sheet.get_all_values()
        if len(records) <= 1:
            return balances

        for row in records[1:]:
            if len(row) < 6:
                continue

            transaction_type = row[1].strip()
            currency = row[5].strip()

            try:
                amount = float(row[4].replace(",", "").strip())
            except Exception:
                continue

            if transaction_type in ["إنتاج", "آجل / دين"]:
                continue

            if currency in balances:
                balances[currency] += amount
    except Exception as e:
        print(f"Error calculate_balances: {e}")

    return balances

def get_balance_after_transaction(currency, amount, transaction_type):
    balances = calculate_balances()
    current_balance = balances.get(currency, 0.0)
    if transaction_type in ["إنتاج", "آجل / دين"]:
        return current_balance
    return current_balance + amount

def add_transaction(
    transaction_type,
    category,
    details,
    amount=0.0,
    currency="ريال يمني",
    operation="",
    explanation="",
    user_name="مستخدم"
):
    date_string = get_date_string()
    transaction_id = generate_transaction_id()
    balance = get_balance_after_transaction(currency, amount, transaction_type)

    row = [
        date_string,
        transaction_type,
        category,
        details,
        amount,
        currency,
        operation,
        balance,
        explanation,
        transaction_id,
        user_name
    ]

    if sheet:
        try:
            sheet.append_row(row, value_input_option="USER_ENTERED")
        except Exception as e:
            print(f"Error append_row: {e}")

    return {
        "date": date_string,
        "id": transaction_id,
        "balance": balance
    }

def analyze_message(text):
    lower_text = convert_arabic_numbers(text).lower()
    currency = detect_currency(text)
    quantity = extract_quantity(text)

    if any(word in lower_text for word in ["انتاج", "إنتاج", "حصد", "حصاد"]):
        if quantity <= 0:
            quantity = extract_amount(text)
        return {
            "type": "إنتاج",
            "category": "إنتاج محصول الخيار",
            "amount": quantity,
            "currency": "كيلو",
            "operation": "إنتاج",
            "quantity": quantity,
            "details": text,
            "explanation": f"تم تسجيل إنتاج {quantity} كيلو خيار"
        }

    amount = extract_amount(text)
    return {
        "type": "مصروف",
        "category": detect_expense_category(text),
        "amount": -abs(amount),
        "currency": currency,
        "operation": "صرف",
        "quantity": 0,
        "details": text,
        "explanation": "مصروف لمشروع زراعة الخيار"
    }

def process_invoice_entry(boxes: float, invoice_net: float, currency: str, user_name: str) -> str:
    packing_cost = boxes * 500.0
    transport_cost = boxes * 500.0
    total_deductions = packing_cost + transport_cost
    final_net = invoice_net - total_deductions

    rev_res = add_transaction("إيراد", "مبيعات خيار", f"فاتورة {boxes:.0f} صندوق", invoice_net, currency, "بيع", "", user_name)
    add_transaction("مصروف", "تغليف وكراتين", f"تكييس {boxes:.0f} صندوق", -packing_cost, currency, "صرف", "", user_name)
    add_transaction("مصروف", "نقل وتسويق", f"نقل {boxes:.0f} صندوق", -transport_cost, currency, "صرف", "", user_name)

    final_balance = rev_res['balance'] - total_deductions

    summary = (
        f"✅ *تم قيد الفاتورة بنجاح*\n"
        f"━━━━━━━━━━━━━━━\n"
        f"📦 عدد الصناديق: *{boxes:,.0f}*\n"
        f"💵 صافي الفاتورة: *{invoice_net:,.0f}* {currency}\n"
        f"✂️ تكلفة التكييس: *{packing_cost:,.0f}* {currency}\n"
        f"🚚 تكلفة النقل: *{transport_cost:,.0f}* {currency}\n"
        f"💰 الصافي النهائي للمزرعة: *{final_net:,.0f}* {currency}\n"
        f"━━━━━━━━━━━━━━━\n"
        f"🏦 رصيد الكاش الحالي: *{final_balance:,.0f}* {currency}"
    )
    return summary

async def handle_exchange(update: Update, context: ContextTypes.DEFAULT_TYPE, text: str):
    norm_text = convert_arabic_numbers(text)
    pattern = r"^صرف\s+(\d+(?:\.\d+)?)\s*(?:سعودي|ر\.س|رس)?\s*بسعر\s*(\d+(?:\.\d+)?)"
    match = re.search(pattern, norm_text, re.IGNORECASE)

    if not match:
        return False

    sar_amount = float(match.group(1))
    rate = float(match.group(2))
    yer_amount = sar_amount * rate

    transaction_id = generate_transaction_id()
    date_string = get_date_string()
    balances = calculate_balances()
    user_name = update.effective_user.first_name or "حسام"

    new_sar_balance = balances["ريال سعودي"] - sar_amount
    new_yer_balance = balances["ريال يمني"] + yer_amount

    if sheet:
        sheet.append_row([
            date_string, "تحويل عملة", "صرافة",
            f"صرف {sar_amount} ريال سعودي بسعر {rate}",
            -sar_amount, "ريال سعودي", "صرافة",
            new_sar_balance, f"تحويل سعودي إلى يمني بسعر {rate}",
            transaction_id, user_name
        ])

        sheet.append_row([
            date_string, "تحويل عملة", "صرافة",
            f"استلام {yer_amount} ريال يمني",
            yer_amount, "ريال يمني", "صرافة",
            new_yer_balance, f"مقابل {sar_amount} ريال سعودي بسعر {rate}",
            transaction_id, user_name
        ])

    reply = (
        f"💱 *صرافة:* {sar_amount:,.0f} ر.س\n"
        f"💵 *المستلم:* {yer_amount:,.0f} ر.ي\n"
        f"📈 *السعر:* {rate:,.0f}\n"
        f"🇸🇦 *رصيد سعودي:* {new_sar_balance:,.0f}\n"
        f"🇾🇪 *رصيد يمني:* {new_yer_balance:,.0f}"
    )

    await update.message.reply_text(reply, parse_mode="Markdown")
    await notify_admins(context, reply, update.effective_chat.id)

    return True

async def handle_message(update: Update, context: ContextTypes.DEFAULT_TYPE):
    if not update.message or not update.message.text:
        return

    raw_text = update.message.text.strip()
    if not raw_text:
        return

    # صرافة
    if raw_text.startswith("صرف"):
        if await handle_exchange(update, context, raw_text):
            return

    # فاتورة سريعة بنص
    if raw_text.startswith("فاتورة") or raw_text.startswith("فاتوره"):
        nums = extract_numbers(raw_text)
        if len(nums) >= 2:
            boxes = nums[0]
            invoice_net = nums[1]
            currency = detect_currency(raw_text)
            user_name = update.effective_user.first_name or "حسام"

            summary = process_invoice_entry(boxes, invoice_net, currency, user_name)
            await update.message.reply_text(summary, parse_mode="Markdown")
            await notify_admins(context, summary, update.effective_chat.id)
            return

    # باقي العمليات (إنتاج أو مصروفات)
    transaction = analyze_message(raw_text)
    if transaction["amount"] == 0 and transaction["type"] != "إنتاج":
        return

    user_name = update.effective_user.first_name or "حسام"
    result = add_transaction(
        transaction_type=transaction["type"],
        category=transaction["category"],
        details=transaction["details"],
        amount=transaction["amount"],
        currency=transaction["currency"],
        operation=transaction["operation"],
        explanation=transaction["explanation"],
        user_name=user_name
    )

    reply = (
        f"🏷️ *النوع:* {transaction['type']}\n"
        f"📋 *الفئة:* {transaction['category']}\n"
        f"💰 *المبلغ / الكمية:* {abs(transaction['amount']):,.0f} {transaction['currency']}\n"
        f"🏦 *الرصيد:* {result['balance']:,.0f} {transaction['currency']}"
    )
    await update.message.reply_text(reply, parse_mode="Markdown")
    await notify_admins(context, reply, update.effective_chat.id)
