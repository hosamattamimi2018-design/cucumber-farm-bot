# =========================================================
# تحويل الأرقام العربية إلى إنجليزية
# =========================================================

def convert_arabic_numbers(text: str) -> str:
    arabic_digits = "٠١٢٣٤٥٦٧٨٩"
    english_digits = "0123456789"

    table = str.maketrans(
        arabic_digits,
        english_digits
    )

    return text.translate(table)


# =========================================================
# تنسيق الأرقام
# =========================================================

def format_number(number):
    try:
        return f"{float(number):,.2f}"
    except Exception:
        return "0.00"


# =========================================================
# التاريخ الحالي
# =========================================================

def get_now():
    return datetime.datetime.now()


def get_date_string():
    return get_now().strftime("%Y-%m-%d %H:%M:%S")


def get_today_string():
    return get_now().strftime("%Y-%m-%d")


# =========================================================
# إنشاء معرف عملية
# =========================================================

def generate_transaction_id():
    now = get_now().strftime("%Y%m%d%H%M%S")
    short_id = uuid.uuid4().hex[:6].upper()

    return f"CUC-{now}-{short_id}"


# =========================================================
# تحديد العملة
# =========================================================

def detect_currency(text):
    text = convert_arabic_numbers(text).lower()

    if any(word in text for word in [
        "سعودي",
        "سعوديه",
        "ريال سعودي",
        "ر.س",
        "رس"
    ]):
        return "ريال سعودي"

    return "ريال يمني"


# =========================================================
# استخراج الأرقام
# =========================================================

def extract_numbers(text):
    text = convert_arabic_numbers(text)

    numbers = re.findall(
        r"\d+(?:\.\d+)?",
        text
    )

    return [float(n) for n in numbers]


def extract_amount(text):
    numbers = extract_numbers(text)

    if not numbers:
        return 0.0

    return numbers[-1]


# =========================================================
# استخراج كمية الخيار
# =========================================================

def extract_quantity(text):
    text = convert_arabic_numbers(text).lower()

    match = re.search(
        r"(\d+(?:\.\d+)?)\s*(?:كيلو|كجم|كغ|kg)",
        text
    )

    if match:
        return float(match.group(1))

    return 0.0


# =========================================================
# استخراج السعر
# =========================================================

def extract_price(text):
    text = convert_arabic_numbers(text).lower()

    patterns = [
        r"بسعر\s*(\d+(?:\.\d+)?)",
        r"سعر\s*(\d+(?:\.\d+)?)",
        r"السعر\s*(\d+(?:\.\d+)?)"
    ]

    for pattern in patterns:
        match = re.search(pattern, text)

        if match:
            return float(match.group(1))

    return 0.0


# =========================================================
# حساب الرصيد
# =========================================================

def calculate_balances():
    records = sheet.get_all_values()

    balances = {
        "ريال سعودي": 0.0,
        "ريال يمني": 0.0
    }

    if len(records) <= 1:
        return balances

    for row in records[1:]:

        if len(row) < 6:
            continue

        transaction_type = row[1].strip()
        currency = row[5].strip()

        try:
            amount = float(
                row[4]
                .replace(",", "")
                .strip()
            )
        except Exception:
            continue

        # الإنتاج ليس حركة مالية
        if transaction_type == "إنتاج":
            continue

        # الآجل لا يدخل في الكاش
        if transaction_type == "آجل / دين":
            continue

        if currency in balances:
            balances[currency] += amount

    return balances


# =========================================================
# الحصول على الرصيد بعد العملية
# =========================================================

def get_balance_after_transaction(currency, amount, transaction_type):
    balances = calculate_balances()

    current_balance = balances.get(currency, 0.0)

    if transaction_type in ["إنتاج", "آجل / دين"]:
        return current_balance

    return current_balance + amount


# =========================================================
# تسجيل عملية مالية أو إنتاج
# =========================================================

def add_transaction(
    transaction_type,
    category,
    details,
    amount=0.0,
    currency="ريال يمني",
    operation="",
    explanation=""
):

    date_string = get_date_string()
    transaction_id = generate_transaction_id()

    balance = get_balance_after_transaction(
        currency,
        amount,
        transaction_type
    )

    row = [
        date_string,
        transaction_type,
        category,
        details,
        amount,
        currency,
        operation,
        balance,
        explanation,
        transaction_id
    ]

    sheet.append_row(
        row,
        value_input_option="USER_ENTERED"
    )

    return {
        "date": date_string,
        "id": transaction_id,
        "balance": balance
    }

# تحليل الرسالة تلقائياً
# =========================================================

def analyze_message(text):

    lower_text = convert_arabic_numbers(text).lower()

    currency = detect_currency(text)

    quantity = extract_quantity(text)
    price = extract_price(text)

    # تسجيل الإنتاج
    if any(word in lower_text for word in [
        "انتاج",
        "إنتاج",
        "حصد",
        "حصاد"
    ]):

        if quantity <= 0:
            quantity = extract_amount(text)

        return {
            "type": "إنتاج",
            "category": "إنتاج محصول الخيار",
            "amount": quantity,
            "currency": "كيلو",
            "operation": "إنتاج",
            "quantity": quantity,
            "details": text,
            "explanation": f"تم تسجيل إنتاج {quantity} كيلو خيار"
        }

    # بيع محصول الخيار
    if any(word in lower_text for word in [
        "بيع",
        "بعت",
        "مبيعات",
        "ايراد",
        "إيراد",
        "دخل",
        "محصول"
    ]):

        if quantity > 0 and price > 0:
            amount = quantity * price

            explanation = (
                f"بيع {quantity:,.2f} كيلو خيار "
                f"بسعر {price:,.2f} "
                f"للكيلو"
            )

        else:
            amount = extract_amount(text)

            explanation = "إيراد من بيع محصول الخيار"

        return {
            "type": "إيراد",
            "category": "مبيعات محصول الخيار",
            "amount": abs(amount),
            "currency": currency,
            "operation": "بيع",
            "quantity": quantity,
            "details": text,
            "explanation": explanation
        }

    # دين / آجل
    if any(word in lower_text for word in [
        "دين",
        "آجل",
        "اجل",
        "مؤجل",
        "مؤجلة"
    ]):

        amount = extract_amount(text)

        return {
            "type": "آجل / دين",
            "category": "ذمم وديون",
            "amount": abs(amount),
            "currency": currency,
            "operation": "آجل",
            "quantity": 0,
            "details": text,
            "explanation": "عملية آجل أو دين"
        }

    # مصروف
    amount = extract_amount(text)

    return {
        "type": "مصروف",
        "category": detect_expense_category(text),
        "amount": -abs(amount),
        "currency": currency,
        "operation": "صرف",
        "quantity": 0,
        "details": text,
        "explanation": "مصروف لمشروع زراعة الخيار"
    }

# معالجة الصرافة
# =========================================================

async def handle_exchange(
    update,
    text
):

    norm_text = convert_arabic_numbers(text)

    pattern = (
        r"^صرف\s+"
        r"(\d+(?:\.\d+)?)"
        r"\s*(?:سعودي|ر\.س|رس)?"
        r"\s*بسعر\s*"
        r"(\d+(?:\.\d+)?)"
    )

    match = re.search(
        pattern,
        norm_text,
        re.IGNORECASE
    )

    if not match:
        return False

    sar_amount = float(match.group(1))
    rate = float(match.group(2))
    yer_amount = sar_amount * rate

    transaction_id = generate_transaction_id()
    date_string = get_date_string()

    balances = calculate_balances()

    new_sar_balance = (
        balances["ريال سعودي"] - sar_amount
    )

    new_yer_balance = (
        balances["ريال يمني"] + yer_amount
    )

    # خصم السعودي
    sheet.append_row([
        date_string,
        "تحويل عملة",
        "صرافة",
        f"صرف {sar_amount} ريال سعودي بسعر {rate}",
        -sar_amount,
        "ريال سعودي",
        "صرافة",
        new_sar_balance,
        f"تحويل سعودي إلى يمني بسعر {rate}",
        transaction_id
    ])

    # إضافة اليمني
    sheet.append_row([
        date_string,
        "تحويل عملة",
        "صرافة",
        f"استلام {yer_amount} ريال يمني",
        yer_amount,
        "ريال يمني",
        "صرافة",
        new_yer_balance,
        f"مقابل {sar_amount} ريال سعودي بسعر {rate}",
        transaction_id
    ])

    await update.message.reply_text(
        "✅ *تمت عملية الصرافة بنجاح*\n\n"

        f"🔻 تم خصم: {format_number(sar_amount)} ر.س\n"
        f"🔺 تم إضافة: {format_number(yer_amount)} ر.ي\n"
        f"💱 سعر الصرف: {format_number(rate)}\n"
        f"🆔 العملية: `{transaction_id}`",

        parse_mode="Markdown"
    )

    return True


# =========================================================
# استقبال الرسائل
# =========================================================

async def handle_message(
    update: Update,
    context: ContextTypes.DEFAULT_TYPE
):

    if not update.message or not update.message.text:
        return

    raw_text = update.message.text.strip()

    if not raw_text:
        return

    # الصرافة
    if raw_text.startswith("صرف"):
        exchange_done = await handle_exchange(
            update,
            raw_text
        )

        if exchange_done:
            return

    # تحليل العملية
    transaction = analyze_message(raw_text)

    # التأكد من وجود مبلغ
    if (
        transaction["amount"] == 0
        and transaction["type"] != "إنتاج"
    ):
        await update.message.reply_text(
            "⚠️ *لم أتمكن من معرفة المبلغ.*\n\n"
            "مثال:\n"
            "`سماد 25000 يمني`\n"
            "`بيع خيار 500 كيلو بسعر 120 سعودي`",

            parse_mode="Markdown"
        )

        return

    # تسجيل العملية
    result = add_transaction(
        transaction_type=transaction["type"],
        category=transaction["category"],
        details=transaction["details"],
        amount=transaction["amount"],
        currency=transaction["currency"],
        operation=transaction["operation"],
        explanation=transaction["explanation"]
    )

    icons = {
        "إيراد": "🟢",
        "مصروف": "🔴",
        "إنتاج": "🥒",
        "آجل / دين": "🟡"
    }

    icon = icons.get(
        transaction["type"],
        "✅"
    )

    # رسالة الإنتاج
    if transaction["type"] == "إنتاج":

        reply = (
            f"{icon} *تم تسجيل الإنتاج بنجاح*\n\n"

            f"📅 التاريخ: {result['date']}\n"
            f"⚖️ الكمية: {format_number(transaction['amount'])} كيلو\n"
            f"📂 الفئة: {transaction['category']}\n"
            f"📝 البيان: {raw_text}\n"
            f"🆔 المعرف: `{result['id']}`"
        )

    # الرسائل المالية
    else:

        reply = (
            f"{icon} *تم تسجيل العملية بنجاح*\n\n"

            f"📅 التاريخ: {result['date']}\n"
            f"🏷️ النوع: {transaction['type']}\n"
            f"📂 الفئة: {transaction['category']}\n"
            f"💰 المبلغ: "
            f"{format_number(abs(transaction['amount']))} "
            f"{transaction['currency']}\n"
        )

        if transaction["quantity"] > 0:
            reply += (
                f"⚖️ الكمية: "
                f"{format_number(transaction['quantity'])} كيلو\n"
            )

        reply += (
            f"🏦 الرصيد: "
            f"{format_number(result['balance'])} "
            f"{transaction['currency']}\n"

            f"📝 البيان: {raw_text}\n"
            f"🆔 المعرف: `{result['id']}`"
        )

    await update.message.reply_text(
        reply,
        parse_mode="Markdown"
    )

def get_today_report():

    records = sheet.get_all_values()

    today = get_today_string()

    production = 0.0

    sar_income = 0.0
    sar_expense = 0.0

    yer_income = 0.0
    yer_expense = 0.0

    transactions_count = 0

    for row in records[1:]:

        if len(row) < 6:
            continue

        if not row[0].startswith(today):
            continue

        transactions_count += 1

        transaction_type = row[1]

        try:
            amount = float(
                row[4].replace(",", "")
            )
        except Exception:
            continue

        currency = row[5]

        if transaction_type == "إنتاج":
            production += abs(amount)
            continue

        if currency == "ريال سعودي":

            if transaction_type == "إيراد":
                sar_income += abs(amount)

            elif transaction_type == "مصروف":
                sar_expense += abs(amount)

        elif currency == "ريال يمني":

            if transaction_type == "إيراد":
                yer_income += abs(amount)

            elif transaction_type == "مصروف":
                yer_expense += abs(amount)

    return {
        "production": production,

        "sar_income": sar_income,
        "sar_expense": sar_expense,

        "yer_income": yer_income,
        "yer_expense": yer_expense,

        "count": transactions_count
    }

# الأرباح الكلية
# =========================================================

def get_profit_report():

    records = sheet.get_all_values()

    sar_income = 0.0
    sar_expense = 0.0

    yer_income = 0.0
    yer_expense = 0.0

    total_production = 0.0

    for row in records[1:]:

        if len(row) < 6:
            continue

        transaction_type = row[1]

        try:
            amount = float(
                row[4].replace(",", "")
            )
        except Exception:
            continue

        currency = row[5]

        if transaction_type == "إنتاج":
            total_production += abs(amount)
            continue

        if transaction_type == "إيراد":

            if currency == "ريال سعودي":
                sar_income += abs(amount)

            elif currency == "ريال يمني":
                yer_income += abs(amount)

        elif transaction_type == "مصروف":

            if currency == "ريال سعودي":
                sar_expense += abs(amount)

            elif currency == "ريال يمني":
                yer_expense += abs(amount)

    return {
        "production": total_production,

        "sar_income": sar_income,
        "sar_expense": sar_expense,

        "sar_profit": sar_income - sar_expense,

        "yer_income": yer_income,
        "yer_expense": yer_expense,

        "yer_profit": yer_income - yer_expense
    }


# =========================================================
# الديون
# =========================================================

def get_debts():

    records = sheet.get_all_values()

    sar_debt = 0.0
    yer_debt = 0.0

    for row in records[1:]:

        if len(row) < 6:
            continue

        if row[1] != "آجل / دين":
            continue

        try:
            amount = abs(
                float(row[4].replace(",", ""))
            )
        except Exception:
            continue

        if row[5] == "ريال سعودي":
            sar_debt += amount

        elif row[5] == "ريال يمني":
            yer_debt += amount

    return sar_debt, yer_debt
